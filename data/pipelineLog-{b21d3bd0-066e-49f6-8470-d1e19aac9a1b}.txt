+ ( set +e
  #=====================
  # IMAGES FIRST, with the CLASSIC builder. The client here is
  # 20.10.24 — old enough that `docker build` is still the classic
  # command and NOT an alias for buildx — so it talks straight to
  # the daemon's builder with no privileged helper container. The
  # tags are exactly the `image:` names in docker-compose.yml, so
  # the later `up`/`run` find them already present and never try to
  # build (compose only builds when the image is missing).
  #
  # bff is absent on purpose: it shares econsent-bff:dev.
  #=====================
  cd /tmp/ctx
  DF=econsent_bff/docker/econsent.Dockerfile
  #=====================
  # Explicit, not inherited: today's client (20.10) defaults to the
  # classic builder anyway, but a future runner image with a 23+
  # client would alias `docker build` to buildx and land straight
  # back on the privileged-container refusal. Pinning it to 0 keeps
  # that upgrade from silently breaking this step.
  #=====================
  # ONE SIDE EFFECT OF THE CLASSIC BUILDER worth knowing: it reads a
  # .dockerignore only from the CONTEXT ROOT, so
  # docker/econsent.Dockerfile.dockerignore — a per-Dockerfile
  # BuildKit feature — is INERT here. That is what lets the CI
  # Dockerfile below COPY docker/env/*.env and docker/seed-ci/, both
  # of which that ignore-file excludes. It is safe precisely because
  # this path is CI-only and those values are the fake set; a
  # developer's `compose build` goes through BuildKit, where the
  # ignore-file applies and their REAL .env never enters a context.
  export DOCKER_BUILDKIT=0
  DFCI=econsent_bff/docker/econsent-ci.Dockerfile
  COMPOSE="docker compose --env-file .env.ci -f docker-compose.yml -f docker-compose.ci.yml"
  docker build --target api        -t econsent-api:dev        -f "$DF" . \
    && docker build --target bff        -t econsent-bff:dev        -f "$DF" . \
    && docker build --target testrunner -t econsent-testrunner:dev -f "$DF" . \
